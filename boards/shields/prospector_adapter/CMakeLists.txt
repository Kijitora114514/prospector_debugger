if(CONFIG_SHIELD_PROSPECTOR_ADAPTER)
  set(PROSPECTOR_SHIELD_DIR ${CMAKE_CURRENT_LIST_DIR})
  zephyr_library()
  file(GLOB font_sources ${PROSPECTOR_SHIELD_DIR}/src/fonts/*.c)
  zephyr_library_sources(${ZEPHYR_BASE}/misc/empty_file.c)
  zephyr_library_include_directories(${ZEPHYR_LVGL_MODULE_DIR})
  zephyr_library_include_directories(${ZEPHYR_BASE}/lib/gui/lvgl/)
  zephyr_library_include_directories(${ZEPHYR_BASE}/drivers)
  zephyr_library_include_directories(${CMAKE_SOURCE_DIR}/include)
  zephyr_library_include_directories(${ZEPHYR_CURRENT_MODULE_DIR}/include)
  zephyr_library_include_directories(${PROSPECTOR_SHIELD_DIR}/include)
  zephyr_library_sources(${PROSPECTOR_SHIELD_DIR}/src/brightness.c)
  zephyr_library_sources(${PROSPECTOR_SHIELD_DIR}/src/custom_status_screen.c)
  zephyr_library_sources(${PROSPECTOR_SHIELD_DIR}/src/display_rotate_init.c)

  if(CONFIG_PROSPECTOR_SHOW_MODIFIERS)
    zephyr_library_sources(${PROSPECTOR_SHIELD_DIR}/src/modifier_order.c)
  endif()

  if(CONFIG_PROSPECTOR_STATUS_SCREEN_DEBUGGER)
    file(GLOB layout_sources ${PROSPECTOR_SHIELD_DIR}/src/layouts/debugger/*.c)
    set(heroicon_sources
      ${PROSPECTOR_SHIELD_DIR}/src/layouts/debugger/lvgl/connect.c
      ${PROSPECTOR_SHIELD_DIR}/src/layouts/debugger/lvgl/disconnect.c
    )
    list(FILTER layout_sources EXCLUDE REGEX "status_screen\\.c$")
    list(FILTER layout_sources EXCLUDE REGEX "output\\.c$")
    list(FILTER layout_sources EXCLUDE REGEX "modifier_indicator\\.c$")
    zephyr_library_sources(${layout_sources} ${heroicon_sources})
  elseif(CONFIG_PROSPECTOR_STATUS_SCREEN_CLASSIC)
    file(GLOB layout_sources ${PROSPECTOR_SHIELD_DIR}/src/layouts/classic/*.c)
    list(FILTER layout_sources EXCLUDE REGEX "status_screen\\.c$")
    zephyr_library_sources(${layout_sources})
  elseif(CONFIG_PROSPECTOR_STATUS_SCREEN_RADII)
    file(GLOB layout_sources ${PROSPECTOR_SHIELD_DIR}/src/layouts/radii/*.c)
    list(FILTER layout_sources EXCLUDE REGEX "status_screen\\.c$")
    zephyr_library_sources(${layout_sources})
  elseif(CONFIG_PROSPECTOR_STATUS_SCREEN_FIELD)
    file(GLOB layout_sources ${PROSPECTOR_SHIELD_DIR}/src/layouts/field/*.c)
    list(FILTER layout_sources EXCLUDE REGEX "status_screen\\.c$")
    zephyr_library_sources(${layout_sources})
  elseif(CONFIG_PROSPECTOR_STATUS_SCREEN_OPERATOR)
    file(GLOB layout_sources ${PROSPECTOR_SHIELD_DIR}/src/layouts/operator/*.c)
    list(FILTER layout_sources EXCLUDE REGEX "status_screen\\.c$")
    zephyr_library_sources(${layout_sources})
  endif()

  zephyr_library_sources(${font_sources})
endif()
